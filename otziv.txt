JS

const dialog = document.getElementById('feedback-dialog');
const openBtn = document.getElementById('open-feedback');
const closeBtn = document.getElementById('close-feedback');
const form = document.getElementById('feedback-form');
const submitBtn = document.querySelector('.submit-feedback-btn');

// Открытие модального окна (нативный метод блокирует страницу под ним)
openBtn.addEventListener('click', () => {
    dialog.showModal();
});

// Закрытие окна по кнопке-крестику
closeBtn.addEventListener('click', () => {
    dialog.close();
});

// Закрытие окна при клике на полупрозрачную подложку (backdrop)
dialog.addEventListener('click', (e) => {
    if (e.target === dialog) {
        dialog.close();
    }
});

// Анимация отправки данных
form.addEventListener('submit', (e) => {
    // Проверка заполненности радио-кнопок
    const ratingSelected = form.querySelector('input[name="rating"]:checked');
    
    if (!ratingSelected) {
        e.preventDefault();
        // Упругая тряска окна, если пользователь забыл выбрать смайлик
        dialog.classList.add('bounce-modal');
        dialog.addEventListener('animationend', () => dialog.classList.remove('bounce-modal'), { once: true });
        return;
    }

    // Имитация успешной отправки на сервер
    e.preventDefault();
    submitBtn.classList.add('sending');
    submitBtn.querySelector('span').textContent = 'Успешно отправлено!';

    // Закрываем окно через 1.5 секунды после демонстрации статуса успеха
    setTimeout(() => {
        dialog.close();
        // Возвращаем кнопку в исходный вид для следующих открытий
        form.reset();
        submitBtn.classList.remove('sending');
        submitBtn.querySelector('span').textContent = 'Отправить отзыв';
    }, 1500);
});
